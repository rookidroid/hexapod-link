# Everything needed to run and build the standalone desktop app.
#
#   python desktop.py            run the native window
#   pyinstaller hexapod.spec     build dist/HexapodLink.exe

-r requirements.txt

# Native window. On Windows this renders through the Edge WebView2 runtime,
# which ships with Windows 10/11; on Linux it uses GTK/WebKit, on macOS WKWebView.
pywebview>=5.0

# The Linux backend is GTK/WebKit2 through PyGObject, which pywebview does not
# pull in itself. Building these from source needs the system headers too:
#
#   sudo apt install libgirepository1.0-dev libcairo2-dev pkg-config \
#                    gir1.2-webkit2-4.1 libwebkit2gtk-4.1-0
#
# PyGObject 3.50 switched to girepository-2.0, which is not in Ubuntu 24.04 or
# earlier; the upper bound keeps the build working on the distributions that
# ship 1.0. Drop it once the target distribution has libgirepository-2.0-dev.
PyGObject>=3.42,<3.50; sys_platform == "linux"
pycairo>=1.23; sys_platform == "linux"

# Production WSGI server. Replaces the Flask dev server that index.py uses,
# which prints warnings and is not meant to be left running.
waitress>=3.0

# Build tooling only; not imported at runtime.
pyinstaller>=6.3
